cmake_minimum_required(VERSION 3.16)
project(exchange_simulator_cpp CXX)

# Default to Release for single-config generators; Debug works via
#   cmake -S . -B build -DCMAKE_BUILD_TYPE=Debug
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "build type" FORCE)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_compile_options(-Wall -Wextra)

# The engine: no dependencies, no I/O — same discipline as orderbook/ in Python.
add_library(orderbook STATIC
  src/order.cpp
  src/order_book.cpp
  src/matching_engine.cpp
)
target_include_directories(orderbook PUBLIC include)

add_executable(orderbook_tests
  tests/main.cpp
  tests/test_price_time.cpp
  tests/test_matching.cpp
  tests/test_cancellation.cpp
)
target_link_libraries(orderbook_tests PRIVATE orderbook)

add_executable(orderbook_benchmark benchmark/main.cpp)
target_link_libraries(orderbook_benchmark PRIVATE orderbook)

# Cross-language differential checker: replays vector files generated by
# `python -m reference.vectors` and must reproduce them exactly.
add_executable(orderbook_differential differential/main.cpp)
target_link_libraries(orderbook_differential PRIVATE orderbook)

enable_testing()
add_test(NAME orderbook_tests COMMAND orderbook_tests)
